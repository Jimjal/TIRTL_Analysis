# Standard Installation (x86_64 / Intel Mac / Windows):
#   conda activate TIRTL_analyse
#   pip install -r requirements.txt
#
# ARM64 (Apple Silicon M1/M2/M3) Installation:
#   The tssv dependency uses x86 SSE intrinsics that don't compile on ARM64.
#   Use the patched version in scripts/patches/tssv-1.1.2-arm64/:
#
#   conda activate TIRTL_analyse
#   pip install --no-deps demultiplex biopython dict-trie fastools jit-open
#   pip install scripts/patches/tssv-1.1.2-arm64/
#
# Note: This project requires the demultiplex tool from jfjlaros
# Documentation: https://demultiplex.readthedocs.io
